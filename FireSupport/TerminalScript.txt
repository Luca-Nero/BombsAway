# =============================================================================
#  BombsAway terminal script
# =============================================================================
#
#  Everything the fire-support terminal types: the commands, the system output,
#  the programs that run each unit, and your own calls. Edit freely. The mod
#  reads this file again whenever it changes, so an edit shows on the next call
#  (no restart). Delete the file to get the default back; a broken line is
#  skipped and named in the MelonLoader log ([Terminal] ...).
#
#  ---------------------------------------------------------------------------
#  LAYOUT
#  ---------------------------------------------------------------------------
#
#  [NAME]              a program (a group). Its settings come first:
#    program = X.AI      its name in the title bar (and {PROGRAM})
#    prefix  = X         written before each of its lines:  X> ...
#    units   = 30, 20    the missions it runs, by their code on the strip:
#                          155 81 SMK ILL PGM   (artillery, ArtyTerminal)
#                          30 20 RKT FLC 500 1K 2K CBU MOAB   (air, AirTerminal)
#                        Two programs with the same unit: one is picked at
#                        random per mission. No program for a unit: that
#                        mission stays on the radio net.
#    rate    = 1.0       its typing speed against RadioTypeRate (1 = yours)
#    pitch   = 1.0       its squelch's pitch (lower = deeper)
#    flicker = off       on: its newest letter flickers through junk as it types
#
#  @step               a step: the lines typed at one moment of the mission.
#                      Steps right under [NAME] are shared by all its
#                      conversations.
#
#  == NAME             a conversation. One is picked per mission (never the
#                      same one twice running). Its steps beat the group's.
#
#  [ALL]               steps every program falls back to.
#
#  Lines inside a step:
#    cmd: ...            a command, typed at the prompt with key clicks
#    sys: ...            system output, dim, almost instant
#    ai: ...             the program speaking (its prefix, its pace, its squelch)
#    obs: ...            you, the observer
#    status: ...         the title bar's state, from the next line on
#
#  A ^ in a line is where it gets killed: it stops there with noise.
#  A step that is there but has no lines says nothing (to silence a fallback).
#  Long lines wrap; the window shows the last 9 rows, 56-58 letters each.
#  Glyphs: A-Z 0-9 and  . , : ; - + / \ < > % # ( ) ? ! _ = * [ ] | " ' { } & $ @
#  Anything else is drawn as a space (and named in the log). Text is upper case.
#
#  ---------------------------------------------------------------------------
#  STEPS, IN ORDER
#  ---------------------------------------------------------------------------
#
#  Every mission starts:   spawn, boot, wake, call   (typed back to back; the
#                          mission's clock starts once the call is typed out)
#  Artillery then:         readback, shot, splash, kill, last, free
#                          (the program is killed at SPLASH, rounds in the air)
#  Air strikes, right after the call:
#                          plan       how it will come in: the attack axis
#                                     picked for the mark, the way in and out
#                          masked     only when the mark is under cover
#                                     (nothing reaches it cleanly)
#  Air strikes then:       readback   about 3 s after the call
#                          inbound    about 11 s before the attack
#                          in         5 s before it (4 s before a release)
#                          away       guns: as they open fire; rockets: last
#                                     pair off; bombs: the release
#                          off        guns and rockets breaking away
#                          impact     bombs only: a second after the impact
#                          end, kill, last, free   once it's all down
#  A call while the unit is busy:   unable  (said by the program on the line)
#
#  Variants: a step is looked up most specific first:
#      step.UNIT.danger   step.UNIT   step.danger   step
#  e.g. @readback.danger (mark close to you), @away.FLC (flechette rockets).
#  Between equally specific steps, the conversation's beats its program's
#  beats [ALL].
#
#  ---------------------------------------------------------------------------
#  WORDS FILLED IN
#  ---------------------------------------------------------------------------
#
#  {GRID}     the mark, 8 figures          {SPAWN}   where the unit was spawned
#  {RANGE}    km from the unit to the mark {AZ}      mils from the unit to it
#  {HDG}      attack heading, 3 figures    {FROM}    "WEST": in from the ...
#  {OFF}      where it breaks off to       {TOF}     a bomb's time of fall, s
#  {DANGER}   "DANGER CLOSE. " or nothing  {SIGN}    call sign: HOG 11
#  {N}        rounds, rockets, bomblets    {NWORD}   the same as a word: SIX
#  {AMMO}     HE, 30MM, FLECHETTE, 2000LB JDAM ...
#  {WEAPON}   155MM HE, 30MM GUN RUN, M151, GBU-31 ...
#  {CRAFT}    M777A2, A-10C, F-22A, AH-64E, F-15E, MC-130J
#  {PROGRAM}  {PREFIX}  {UNIT} (the strip code)  {PID} (a random process id)
#  Air strikes' plan (AirDynamicApproach):
#  {DIVE}     dive, or a bomb's impact angle, degrees
#  {LOS}      % of the lines tried along the axis that reach the mark
#  {AXES}     attack axes tried            {COVER}   surfaces on the axis
#  {ENTRY}    where it comes in from: "SOUTHWEST"
#  {EGRESS}   "BREAK NORTH" or "OVERFLY, OFF NORTH"
#  {FUZE}     a bomb's fuze: INSTANT, or DELAY under cover
#
# =============================================================================


# -----------------------------------------------------------------------------
#  Fallbacks for every program
# -----------------------------------------------------------------------------
[ALL]

@spawn
status: SPAWNING
cmd: SPAWN BA:{CRAFT}
sys: ALLOC 1X {CRAFT} .......... GRID {SPAWN}

@boot
status: BOOTING
cmd: RUN {PROGRAM} /BIND {CRAFT}

@wake
status: LINKED
ai: ONLINE.

@call
obs: {SIGN}, GRID {GRID}. {DANGER}{WEAPON}. OVER.

@unable
obs: {SIGN}, GRID {GRID}. OVER.
ai: UNABLE. ENGAGED. OUT.

@plan
sys: SOLVE ATTACK AXIS ...... {AXES} TRIED, LOS {LOS}%
sys: AXIS {HDG} FROM {FROM} AT {DIVE} DEG, ENTRY {ENTRY}
sys: EGRESS {EGRESS}

@masked
sys: WARN: NO CLEAR LINE TO GRID {GRID} ({COVER} SURFACES)
ai: TARGET IS UNDER COVER. GOING THROUGH IT.

@readback
ai: GRID {GRID}. {WEAPON}. OUT.
obs: READBACK CORRECT.

@shot
status: SHOT
ai: SHOT. OVER.
obs: SHOT. OUT.

@splash
status: SPLASH
ai: SPLASH. OVER.
obs: SPLASH. OUT.

@inbound
status: INBOUND
ai: IP INBOUND.
obs: CONTINUE.

@in
status: HOT
ai: IN FROM THE {FROM}.
obs: CLEARED HOT.

@off
status: OFF
ai: OFF {OFF}.

@end
obs: GOOD EFFECT ON TARGET. END OF MISSION. OUT.

@kill
cmd: KILL {PROGRAM} /F

@free
sys: [PROCESS 0X{PID} KILLED]
cmd: FREE BA:{CRAFT}
sys: 1X {CRAFT} DEALLOCATED
status: TERMINATED
sys: GRID {SPAWN} PURGED
